export function inspect(value: unknown): boolean {
  return value !== null;
}

export const identity = (value: unknown): unknown => value;

declare const textSchema: { parse(value: unknown): string };

export function parseAndLeak(value: unknown): unknown {
  textSchema.parse(value);
  return value;
}

export type Consumer = (value: unknown) => void;

function exportedByList(value: unknown): boolean {
  return value !== undefined;
}
export { exportedByList };

export class Store {
  put(value: unknown): boolean {
    return value !== null;
  }
}

export interface Sink {
  write(value: unknown): void;
}
