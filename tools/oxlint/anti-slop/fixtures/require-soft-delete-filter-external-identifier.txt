import { exists, sql } from "drizzle-orm";
import { db } from "~/server/db/db";
import { entityExternalId } from "~/server/db/schema";

export function missingGuard() {
  return db
    .select()
    .from(entityExternalId)
    .where(exists(db.select().from(entityExternalId).where(sql`${entityExternalId.id} = 1`)));
}
